import React from 'react';
import { Dimensions } from 'react-native';
import Svg, { Defs, RadialGradient, Stop, Rect } from 'react-native-svg';

const { width, height } = Dimensions.get('window');

// Generic atmosphere factory — takes phase-specific color stops
function PhaseAtmosphere({ stops }) {
  return (
    <Svg
      width={width * 2}
      height={height * 0.75}
      style={{ position: 'absolute', top: 0, left: -width * 0.5 }}
      pointerEvents="none"
    >
      <Defs>
        <RadialGradient id={`glow-${stops[0].color}`} cx="50%" cy="20%" rx="60%" ry="55%" fx="50%" fy="20%">
          {stops.map((stop, i) => (
            <Stop key={i} offset={stop.offset} stopColor={stop.color} stopOpacity={stop.opacity} />
          ))}
        </RadialGradient>
      </Defs>
      <Rect x="0" y="0" width="100%" height="100%" fill={`url(#glow-${stops[0].color})`} />
    </Svg>
  );
}

// Phase I — The Signal
export function ArchaicAtmosphere() {
  return <PhaseAtmosphere stops={[
    { offset: '0%',   color: '#3D1E3D', opacity: '0.85' },
    { offset: '35%',  color: '#2A1530', opacity: '0.6' },
    { offset: '65%',  color: '#1A0D1F', opacity: '0.25' },
    { offset: '100%', color: '#050208', opacity: '0' },
  ]} />;
}
